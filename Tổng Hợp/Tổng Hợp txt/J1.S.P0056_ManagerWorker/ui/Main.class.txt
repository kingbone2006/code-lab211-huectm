package ui;

import bo.Manager;
import entity.SalaryStatus;
import entity.Worker;
import utils.Validator;

public class Main {

    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        // TODO code application logic here
        Manager control = new Manager();
        do {
            int choice = Validator.getInt("======== Worker Management =========\n"
                    + "1.  Add Worker\n"
                    + "2.  Up salary\n"
                    + "3.  Down salary\n"
                    + "4.  Display Information salary\n"
                    + "5.  Display list worker\n"
                    + "Enter your choice: ",
                    "Just 1 -> 5", "Invalid!", 1, 5);
            switch (choice) {
                case 1:
                    try {
                    Worker worker = new Worker();
                    worker.input();
                    control.addWorker(worker);
                } catch (Exception ex) {
                    System.out.println(ex.getMessage());
                }
                break;
                case 2:
                    try {
                    String code = Validator.getString("Enter id: ", "Invalid!", "[Ww] \\d+").toUpperCase();
                    double amount = Validator.getDouble("Enter Salary: ", "salary must be > 0", "Invalid!",
                            Double.MIN_VALUE, Double.MAX_VALUE);
                    control.changeSalary(SalaryStatus.UP, code, amount);
                } catch (Exception ex) {
                    System.out.println(ex.getMessage());
                }
                break;
                case 3:
                    try {
                    String code = Validator.getString("Enter id: ", "Invalid!", "[Ww] \\d+").toUpperCase();
                    double amount = Validator.getDouble("Enter Salary: ", "salary must be > 0", "Invalid!",
                            Double.MIN_VALUE, Double.MAX_VALUE);
                    control.changeSalary(SalaryStatus.DOWN, code, amount);
                } catch (Exception ex) {
                    System.out.println(ex.getMessage());
                }
                break;
                case 4:
                    try {
                    control.displaySalaryChangeHistory();
                } catch (Exception ex) {
                    System.out.println(ex.getMessage());
                }
                break;
                case 5:
                    try {
                    control.displayWorkerList();
                } catch (Exception ex) {
                    System.out.println(ex.getMessage());
                }
                break;
            }
        } while (true);
    }

}
