package array;

import java.util.Random;
public class MyArray {
    
    private int[] array;

    /**
     * Constructor: khởi tạo đối tượng MyArray.
     */
    public MyArray(int number) throws Exception {
        if (number <= 0) {
            throw new Exception("number parameter must be > 0");
        }
        this.array = new int[number];
    }

    /**
     * Getter: trả về giá trị thuộc tính Array.
     */
    public int[] getArray() {
        return array;
    }

    /**
     * Setter: gán giá trị cho thuộc tính Array.
     */
    public void setArray(int[] array) {
        this.array = array;
    }

    /**
     * Phương thức genRandomArray: xử lý logic nghiệp vụ của lớp.
     */
    public void genRandomArray() throws Exception {
        Random rand = new Random();
        for (int i = 0; i < array.length; i++) {
            array[i] = rand.nextInt(array.length);
        }
    }

    @Override
    /**
     * Chuyển đối tượng thành chuỗi để hiển thị.
     */
    public String toString() {
        String str = "[";
        for (int i = 0; i < array.length; i++) {
            str += array[i];
            if (i == array.length - 1) {
                str += "]";
            } else {
                str += ", ";
            }
        }
        return str;
    }

    /**
     * Tìm kiếm dữ liệu (searchIndex).
     */
    public int searchIndex(int value) {
        for (int i = 0; i < array.length; i++) {
            if (array[i] == value) {
                return i;
            }
        }
        return -1;
    }
}
