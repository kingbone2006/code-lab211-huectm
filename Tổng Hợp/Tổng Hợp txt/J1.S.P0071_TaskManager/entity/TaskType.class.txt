package entity;

public enum TaskType {
    CODE(1, "Code"),
    TEST(2, "Test"),
    DESIGN(3, "Design"),
    REVIEW(4, "Review");

    private int id;
    private String name;

    // Constructor for the enum
    private TaskType(int id, String name) {
        this.id = id;
        this.name = name;
    }

    /**
     * Getter: trả về giá trị thuộc tính TaskTypeByID.
     */
    public static TaskType getTaskTypeByID(int ID) {
        switch (ID) {
            case 1:
                return CODE;
            case 2:
                return TEST;
            case 3:
                return DESIGN;
            case 4:
                return REVIEW;
            default:
                throw new AssertionError();
        }
    }

    /**
     * Lấy ID.
     */
    public int getId() {
        return id;
    }

    /**
     * Lấy tên.
     */
    public String getName() {
        return name;
    }
}
