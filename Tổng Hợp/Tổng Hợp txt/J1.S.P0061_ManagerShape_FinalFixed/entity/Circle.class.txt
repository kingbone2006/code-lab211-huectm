package entity;

import utils.Validator;
public final class Circle implements Shape {

    private double radius;

    /**
     * Constructor: khởi tạo đối tượng Circle.
     */
    public Circle() {
    }

    /**
     * Constructor: khởi tạo đối tượng Circle.
     */
    public Circle(double radius) throws Exception {
        setRadius(radius);
    }
    
    /**
     * Lấy bán kính.
     */
    public double getRadius() {
        return radius;
    }

    /**
     * Gán bán kính.
     */
    public void setRadius(double radius) throws Exception {
        if (radius > 0) {
            this.radius = radius;
        } else {
            throw new Exception("Radius must be >0");
        }
    }

    @Override
    /**
     * Tính chu vi hình.
     */
    public double getPerimeter()  {
        return 2 * Math.PI * radius;
    }

    @Override
    /**
     * Tính diện tích hình.
     */
    public double getArea() {
        return Math.PI * radius * radius;
    }

    @Override
    /**
     * In kết quả chu vi / diện tích ra màn hình.
     */
    public void printResult() {
        System.out.println("-----Circle-----");
        System.out.println("Radius: " + radius);
        try {
            System.out.println("Area: " + getArea());
            System.out.println("Perimeter: " + getPerimeter());
        } catch (Exception ex) {
            System.out.println(ex.getMessage());
        }
    }
    @Override
    /**
     * Nhập dữ liệu từ bàn phím cho đối tượng.
     */
    public void input() {
        radius=Validator.getDouble("Please input radius of Circle:",
                "Please enter number >0", "Invalid!", Double.MIN_VALUE, Double.MAX_VALUE);
    }
}
