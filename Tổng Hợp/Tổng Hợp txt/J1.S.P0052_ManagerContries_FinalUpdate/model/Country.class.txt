package model;
import utils.Validator;

public class Country {

    protected String countryCode;
    protected String countryName;
    protected double totalArea;

    /**
     * Constructor: khởi tạo đối tượng Country.
     */
    public Country() {
    }

    /**
     * Constructor: khởi tạo đối tượng Country.
     */
    public Country(String countryCode, String countryName, double totalArea) {
        setCountryCode(countryCode);
        setCountryName(countryName);
        setTotalArea(totalArea);
    }

    /**
     * Lấy mã quốc gia.
     */
    public String getCountryCode() {
        return countryCode;
    }

    /**
     * Gán mã quốc gia.
     */
    public void setCountryCode(String countryCode) {
        if (countryCode == null || countryCode.isEmpty() || !countryCode.matches("[A-Za-z0-9]+")) {
            throw new IllegalArgumentException("Invalid country code. It cannot be empty and must contain only letters.");
        }
        this.countryCode = countryCode;
    }

    /**
     * Lấy tên quốc gia.
     */
    public String getCountryName() {
        return countryName;
    }

    /**
     * Gán tên quốc gia.
     */
    public void setCountryName(String countryName) {
        if (countryName == null || countryName.isEmpty() || !countryName.matches("([A-Za-z]+\\s?)+")) {
            throw new IllegalArgumentException("Invalid country name. It cannot be empty and must contain only letters and spaces.");
        }
        this.countryName = countryName;
    }

    /**
     * Lấy tổng diện tích.
     */
    public double getTotalArea() {
        return totalArea;
    }

    /**
     * Gán tổng diện tích.
     */
    public void setTotalArea(double totalArea) {
        if (totalArea <= 0) {
            throw new IllegalArgumentException("Invalid total area. It must be a positive number.");
        }
        this.totalArea = totalArea;
    }

    /**
     * Nhập dữ liệu từ bàn phím cho đối tượng.
     */
    public void input() throws Exception {
        setCountryCode(Validator.getString("Enter code of country:\n",
                "Invalid", "[A-Za-z0-9]+"));
        setCountryName(Validator.getString("Enter name of country:\n",
                "Invalid", "([A-Za-z]+\\s?)+"));
        setTotalArea(Validator.getDouble("Enter total Area:\n",
                "Error range!", "Invalid", 0, Double.MAX_VALUE));
    }

    /**
     * Hiển thị dữ liệu (mảng / danh sách / đối tượng) ra màn hình.
     */
    public void display() {
        System.out.printf("%-10s%-25s%-20.1f\n", getCountryCode(), getCountryName(), totalArea);
    }
}
